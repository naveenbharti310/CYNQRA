"""Dog walk tracker web app. Standard library only.

Run: PORT=8000 DATA_FILE=walks.json python app.py   (listens on 127.0.0.1 only)
"""
from __future__ import annotations

import json
import os
import re
from datetime import date
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from urllib.parse import parse_qs, urlparse

from walks import WalkStore

HERE = os.path.dirname(os.path.abspath(__file__))
WALK_ACTION = re.compile(r"^/api/walks/([A-Za-z0-9_-]+)/(assign|done|undo)$")


def make_server(port: int, data_file: str | None = None) -> ThreadingHTTPServer:
    store = WalkStore(data_file or os.environ.get("DATA_FILE") or "walks.json")

    class Handler(BaseHTTPRequestHandler):
        def log_message(self, *args):
            return

        def _send(self, code, body, kind="application/json; charset=utf-8"):
            data = body if isinstance(body, bytes) else json.dumps(body).encode("utf-8")
            self.send_response(code)
            self.send_header("Content-Type", kind)
            self.send_header("Content-Length", str(len(data)))
            self.send_header("Cache-Control", "no-store")
            self.end_headers()
            self.wfile.write(data)

        def _json(self) -> dict:
            n = int(self.headers.get("Content-Length") or 0)
            data = json.loads(self.rfile.read(n).decode("utf-8")) if n else {}
            if not isinstance(data, dict):
                raise ValueError("send a JSON object")
            return data

        def do_GET(self):
            u = urlparse(self.path)
            q = {k: v[0] for k, v in parse_qs(u.query).items()}
            try:
                if u.path == "/health":
                    return self._send(200, {"ok": True})
                if u.path in ("/", "/index.html"):
                    with open(os.path.join(HERE, "index.html"), "rb") as fh:
                        return self._send(200, fh.read(), "text/html; charset=utf-8")
                if u.path == "/api/walkers":
                    return self._send(200, {"walkers": store.walkers()})
                if u.path == "/api/day":
                    on = q.get("date") or date.today().isoformat()
                    return self._send(200, {"date": on, "walks": store.day(on)})
                if u.path == "/api/today":
                    walker = q.get("walker", "")
                    today = date.today()
                    return self._send(200, {"today": today.isoformat(), "walker": walker,
                                            "walks": store.today_for(walker, today)})
            except ValueError as exc:
                return self._send(400, {"error": str(exc)})
            return self._send(404, {"error": "not found"})

        def do_POST(self):
            path = urlparse(self.path).path
            try:
                body = self._json()
            except ValueError as exc:
                return self._send(400, {"error": f"bad JSON: {exc}"})
            try:
                if path == "/api/walkers":
                    return self._send(201, {"name": store.add_walker(body.get("name")), "walkers": store.walkers()})
                if path == "/api/walkers/remove":
                    try:
                        store.remove_walker(str(body.get("name") or ""))
                    except KeyError:
                        return self._send(404, {"error": "no such walker"})
                    return self._send(200, {"walkers": store.walkers()})
                if path == "/api/walks":
                    return self._send(201, store.add_walk(body.get("date"), body.get("time"), body.get("dog"),
                                                          body.get("details", ""), body.get("walker", "")))
                m = WALK_ACTION.match(path)
                if m:
                    walk_id, action = m.groups()
                    try:
                        if action == "assign":
                            walk = store.assign(walk_id, body.get("walker", ""))
                        elif action == "done":
                            walk = store.mark_done(walk_id, body.get("note"))
                        else:
                            walk = store.undo_done(walk_id)
                    except KeyError:
                        return self._send(404, {"error": f"no walk {walk_id}"})
                    return self._send(200, walk)
            except ValueError as exc:
                return self._send(400, {"error": str(exc)})
            return self._send(404, {"error": "not found"})

    return ThreadingHTTPServer(("127.0.0.1", port), Handler)


if __name__ == "__main__":
    server = make_server(int(os.environ.get("PORT", "8000")))
    print(f"Walk tracker on http://127.0.0.1:{server.server_address[1]}", flush=True)
    try:
        server.serve_forever()
    except KeyboardInterrupt:
        pass

import threading

from app import make_server


class AppTests(unittest.TestCase):
    def setUp(self):
        self.dir = tempfile.TemporaryDirectory()
        self.data = os.path.join(self.dir.name, "walks.json")
        self.start()

    def start(self):
        self.server = make_server(0, self.data)
        self.base = f"http://127.0.0.1:{self.server.server_address[1]}"
        threading.Thread(target=self.server.serve_forever, daemon=True).start()

    def stop(self):
        self.server.shutdown()
        self.server.server_close()

    def tearDown(self):
        self.stop()
        self.dir.cleanup()

    def get(self, path):
        try:
            with urllib.request.urlopen(self.base + path, timeout=5) as r:
                body = r.read()
                return r.status, (json.loads(body) if r.headers.get_content_type() == "application/json" else body)
        except urllib.error.HTTPError as exc:
            return exc.code, json.loads(exc.read())

    def post(self, path, body=None):
        raw = body if isinstance(body, bytes) else json.dumps(body or {}).encode()
        req = urllib.request.Request(self.base + path, data=raw, method="POST", headers={"Content-Type": "application/json"})
        try:
            with urllib.request.urlopen(req, timeout=5) as r:
                return r.status, json.loads(r.read())
        except urllib.error.HTTPError as exc:
            return exc.code, json.loads(exc.read())

    def today(self, n=0):
        return (date.today() + timedelta(days=n)).isoformat()

"""Walk store for the dog walk tracker. Standard library only, one JSON file, atomic writes."""
from __future__ import annotations

import json
import os
import re
import tempfile
import threading
import uuid
from datetime import date, datetime

TIME = re.compile(r"^([01]\d|2[0-3]):[0-5]\d$")
NOTE_MAX = 280


def _date(value) -> str:
    text = str(value or "").strip()
    try:
        return datetime.strptime(text, "%Y-%m-%d").date().isoformat()
    except ValueError:
        raise ValueError(f"date {text!r} must be a real date as YYYY-MM-DD") from None


class WalkStore:
    def __init__(self, path):
        self.path = str(path)
        self._lock = threading.RLock()
        self._walkers: list[str] = []
        self._walks: dict[str, dict] = {}
        if os.path.exists(self.path):
            with open(self.path, encoding="utf-8") as fh:
                text = fh.read().strip()
            data = json.loads(text) if text else {}
            self._walkers = list(data.get("walkers", []))
            self._walks = {w["id"]: w for w in data.get("walks", [])}

    def _save(self) -> None:
        folder = os.path.dirname(os.path.abspath(self.path))
        os.makedirs(folder, exist_ok=True)
        fd, tmp = tempfile.mkstemp(prefix=".walks_", suffix=".json", dir=folder)
        try:
            with os.fdopen(fd, "w", encoding="utf-8") as fh:
                json.dump({"walkers": self._walkers, "walks": list(self._walks.values())}, fh, indent=1)
            os.replace(tmp, self.path)
        except BaseException:
            if os.path.exists(tmp):
                os.unlink(tmp)
            raise

    # walkers ------------------------------------------------------------
    def walkers(self) -> list[str]:
        return sorted(self._walkers, key=str.lower)

    def add_walker(self, name) -> str:
        name = str(name or "").strip()
        if not name:
            raise ValueError("walker name is required")
        with self._lock:
            if name.lower() in (w.lower() for w in self._walkers):
                raise ValueError(f"{name} is already on the walker list")
            self._walkers.append(name)
            self._save()
        return name

    def remove_walker(self, name) -> None:
        with self._lock:
            if name not in self._walkers:
                raise KeyError(name)
            self._walkers.remove(name)
            for w in self._walks.values():
                if w["walker"] == name:
                    w["walker"] = ""
            self._save()

    def _check_walker(self, walker) -> str:
        walker = str(walker or "").strip()
        if walker and walker not in self._walkers:
            raise ValueError(f"{walker} is not on the walker list")
        return walker

    # walks --------------------------------------------------------------
    def add_walk(self, date, time, dog, details="", walker="") -> dict:
        d = _date(date)
        t = str(time or "").strip()
        if not TIME.match(t):
            raise ValueError(f"time {t!r} must be HH:MM in 24 hours")
        dog = str(dog or "").strip()
        if not dog:
            raise ValueError("dog name is required")
        with self._lock:
            walk = {"id": uuid.uuid4().hex[:8], "date": d, "time": t, "dog": dog,
                    "details": str(details or "").strip(), "walker": self._check_walker(walker),
                    "done": False, "note": "", "done_at": ""}
            self._walks[walk["id"]] = walk
            self._save()
        return dict(walk)

    def _get(self, walk_id) -> dict:
        if walk_id not in self._walks:
            raise KeyError(walk_id)
        return self._walks[walk_id]

    def get(self, walk_id) -> dict:
        return dict(self._get(walk_id))

    def assign(self, walk_id, walker) -> dict:
        with self._lock:
            w = self._get(walk_id)
            w["walker"] = self._check_walker(walker)
            self._save()
            return dict(w)

    def day(self, on) -> list[dict]:
        d = _date(on)
        return [dict(w) for w in sorted(self._walks.values(), key=lambda w: (w["time"], w["dog"])) if w["date"] == d]

    def today_for(self, walker, today) -> list[dict]:
        d = today.isoformat() if isinstance(today, date) else _date(today)
        return [w for w in self.day(d) if walker and w["walker"] == walker]

    def mark_done(self, walk_id, note, now=None) -> dict:
        note = str(note or "").strip()
        if not note:
            raise ValueError("add a short note when marking a walk done")
        if len(note) > NOTE_MAX:
            raise ValueError(f"the note is {len(note)} characters; keep it to {NOTE_MAX}")
        with self._lock:
            w = self._get(walk_id)
            w.update({"done": True, "note": note,
                      "done_at": (now or datetime.now()).isoformat(timespec="minutes")})
            self._save()
            return dict(w)

    def undo_done(self, walk_id) -> dict:
        with self._lock:
            w = self._get(walk_id)
            w.update({"done": False, "note": "", "done_at": ""})
            self._save()
            return dict(w)
