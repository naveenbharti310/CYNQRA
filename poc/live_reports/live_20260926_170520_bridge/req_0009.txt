"""Custom cake order tracker web app. Standard library only.

Run: PORT=8000 DATA_FILE=orders.json python app.py   (listens on 127.0.0.1 only)
"""
from __future__ import annotations

import json
import os
import re
from datetime import date
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer

from store import OrderStore

HERE = os.path.dirname(os.path.abspath(__file__))
TOGGLE = re.compile(r"^/api/orders/([A-Za-z0-9_-]+)/(paid|picked_up)$")


def make_server(port: int, data_file: str | None = None) -> ThreadingHTTPServer:
    store = OrderStore(data_file or os.environ.get("DATA_FILE") or "orders.json")

    class Handler(BaseHTTPRequestHandler):
        def log_message(self, *args):
            return

        def _send(self, code: int, body, kind: str = "application/json; charset=utf-8"):
            data = body if isinstance(body, bytes) else json.dumps(body).encode("utf-8")
            self.send_response(code)
            self.send_header("Content-Type", kind)
            self.send_header("Content-Length", str(len(data)))
            self.send_header("Cache-Control", "no-store")
            self.end_headers()
            self.wfile.write(data)

        def _json(self) -> dict:
            n = int(self.headers.get("Content-Length") or 0)
            data = json.loads(self.rfile.read(n).decode("utf-8") or "{}") if n else {}
            if not isinstance(data, dict):
                raise ValueError("send a JSON object")
            return data

        def do_GET(self):
            path = self.path.split("?", 1)[0]
            if path == "/health":
                return self._send(200, {"ok": True})
            if path in ("/", "/index.html"):
                with open(os.path.join(HERE, "index.html"), "rb") as fh:
                    return self._send(200, fh.read(), "text/html; charset=utf-8")
            if path == "/api/orders":
                today = date.today()
                return self._send(200, {"today": today.isoformat(), "due": store.due_soon(today),
                                        "open": store.open_orders()})
            return self._send(404, {"error": "not found"})

        def do_POST(self):
            path = self.path.split("?", 1)[0]
            try:
                body = self._json()
            except ValueError as exc:
                return self._send(400, {"error": f"bad JSON: {exc}"})
            if path == "/api/orders":
                try:
                    order = store.add(body.get("customer"), body.get("cake"), body.get("pickup_date"),
                                      body.get("phone", ""), body.get("notes", ""))
                except ValueError as exc:
                    return self._send(400, {"error": str(exc)})
                return self._send(201, order)
            m = TOGGLE.match(path)
            if m:
                order_id, field = m.groups()
                value = bool(body.get(field))
                try:
                    order = store.set_paid(order_id, value) if field == "paid" else store.set_picked_up(order_id, value)
                except KeyError:
                    return self._send(404, {"error": f"no order {order_id}"})
                return self._send(200, order)
            return self._send(404, {"error": "not found"})

    return ThreadingHTTPServer(("127.0.0.1", port), Handler)


if __name__ == "__main__":
    server = make_server(int(os.environ.get("PORT", "8000")))
    print(f"Order tracker on http://127.0.0.1:{server.server_address[1]}", flush=True)
    try:
        server.serve_forever()
    except KeyboardInterrupt:
        pass

"""Custom cake order store. Standard library only, one JSON file, atomic writes."""
from __future__ import annotations

import json
import os
import tempfile
import threading
import uuid
from datetime import date, datetime, timedelta

DUE_WINDOW_DAYS = 3  # decided rule t_02: today through today plus 3 days, today included
FIELDS = ("id", "customer", "cake", "pickup_date", "phone", "notes", "paid", "picked_up", "created_at")


def parse_date(value) -> date:
    text = str(value or "").strip()
    if not text:
        raise ValueError("pickup date is required, as YYYY-MM-DD")
    try:
        return datetime.strptime(text, "%Y-%m-%d").date()
    except ValueError:
        raise ValueError(f"pickup date {text!r} is not a real date in the form YYYY-MM-DD") from None


class OrderStore:
    def __init__(self, path):
        self.path = str(path)
        self._lock = threading.Lock()
        self._orders: dict[str, dict] = {}
        if os.path.exists(self.path):
            with open(self.path, encoding="utf-8") as fh:
                text = fh.read().strip()
            data = json.loads(text) if text else []
            for o in data:
                self._orders[o["id"]] = o

    def _save(self) -> None:
        folder = os.path.dirname(os.path.abspath(self.path))
        os.makedirs(folder, exist_ok=True)
        fd, tmp = tempfile.mkstemp(prefix=".orders_", suffix=".json", dir=folder)
        try:
            with os.fdopen(fd, "w", encoding="utf-8") as fh:
                json.dump(list(self._orders.values()), fh, indent=1)
            os.replace(tmp, self.path)
        except BaseException:
            if os.path.exists(tmp):
                os.unlink(tmp)
            raise

    def add(self, customer, cake, pickup_date, phone="", notes="") -> dict:
        customer = str(customer or "").strip()
        cake = str(cake or "").strip()
        if not customer:
            raise ValueError("customer is required")
        if not cake:
            raise ValueError("cake is required: say what to make")
        when = parse_date(pickup_date)
        order = {
            "id": uuid.uuid4().hex[:8],
            "customer": customer,
            "cake": cake,
            "pickup_date": when.isoformat(),
            "phone": str(phone or "").strip(),
            "notes": str(notes or "").strip(),
            "paid": False,
            "picked_up": False,
            "created_at": datetime.now().isoformat(timespec="seconds"),
        }
        with self._lock:
            self._orders[order["id"]] = order
            self._save()
        return dict(order)

    def get(self, order_id) -> dict:
        if order_id not in self._orders:
            raise KeyError(order_id)
        return dict(self._orders[order_id])

    def all(self) -> list[dict]:
        return [dict(o) for o in sorted(self._orders.values(), key=lambda o: (o["pickup_date"], o["created_at"]))]

    def _set(self, order_id, key, value: bool) -> dict:
        with self._lock:
            if order_id not in self._orders:
                raise KeyError(order_id)
            self._orders[order_id][key] = bool(value)
            self._save()
            return dict(self._orders[order_id])

    def set_paid(self, order_id, paid: bool) -> dict:
        return self._set(order_id, "paid", paid)

    def set_picked_up(self, order_id, picked_up: bool) -> dict:
        return self._set(order_id, "picked_up", picked_up)

    def open_orders(self) -> list[dict]:
        return [o for o in self.all() if not o["picked_up"]]

    def due_soon(self, today: date) -> list[dict]:
        """Overdue first, then pickup dates from today through today plus DUE_WINDOW_DAYS, soonest first."""
        last = today + timedelta(days=DUE_WINDOW_DAYS)
        overdue, due = [], []
        for o in self.open_orders():
            d = date.fromisoformat(o["pickup_date"])
            if d < today:
                overdue.append(dict(o, overdue=True))
            elif d <= last:
                due.append(dict(o, overdue=False))
        return overdue + due

        self.server.shutdown()
        self.server.server_close()
        self.dir.cleanup()

    def get(self, path):
        try:
            with urllib.request.urlopen(self.base + path, timeout=5) as r:
                return r.status, r.read()
        except urllib.error.HTTPError as exc:
            return exc.code, exc.read()

    def post(self, path, body):
        raw = body if isinstance(body, bytes) else json.dumps(body).encode()
        req = urllib.request.Request(self.base + path, data=raw, method="POST", headers={"Content-Type": "application/json"})
        try:
            with urllib.request.urlopen(req, timeout=5) as r:
                return r.status, json.loads(r.read())
        except urllib.error.HTTPError as exc:
            return exc.code, json.loads(exc.read())

    def add(self, days=1, **kw):
        body = {"customer": "Priya", "cake": "Chocolate", "pickup_date": (date.today() + timedelta(days=days)).isoformat()}
